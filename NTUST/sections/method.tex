\chapter{Proposed Method}
\label{sec:method}

This chapter details the proposed dynamic timing adjustment algorithm. To address the unstable transmission latency during packet transmission between the PNF and VNF, we propose an Adaptive Slots Ahead mechanism.

\begin{enumerate}
    \item \textcolor{blue}{\textbf{Inner-Loop (Node Sync):} Maintains phase and slot boundary alignment between the VNF and PNF. It periodically exchanges timestamps to calculate the absolute phase offset, correcting for hardware clock drift (Clock Drift).}
    \item \textcolor{blue}{\textbf{Outer-Loop (Delay Management):} Operates on top of the stabilized phase reference provided by the inner loop. It tracks and estimates packet transport delays and jitter at the slot level, dynamically adjusting the slots ahead ($S_{ahead}$) to guarantee that control messages fall within the PNF's strict timing window.}
\end{enumerate}

\section{Inner-Loop: Node Sync}
\label{subsec:node_sync}
\textcolor{blue}{To perform software-level clock synchronization, the system adopts a timestamp exchange mechanism similar to the Network Time Protocol (NTP). The VNF periodically transmits a downlink node synchronization message with a local timestamp $t_1$. Upon receiving the message, the PNF records the local arrival timestamp $t_2$ and transmits an uplink node synchronization message back to the VNF, including the transmit timestamp $t_3$ and the echoed $t_2$. When the VNF receives the message, it records its local arrival timestamp $t_4$. Because the nFAPI timestamps are constrained by a 1024 subframe number (SFN) loop, they wrap around every $10.24$~seconds ($10240000$~\textmu s). The algorithm calculates the differences $d_1 = t_2 - t_1$ and $d_2 = t_4 - t_3$, applying a wrap-around correction to align them within the interval $[-5120000, 5120000]$~\textmu s. The VNF then calculates the instantaneous clock offset:}
\begin{equation}
\mathit{offset} = \frac{d_1 - d_2}{2}
\end{equation}
\textcolor{blue}{To prevent timing adjustments from causing control loop oscillations, the synchronization control loop implements a state-based dual-threshold hysteresis mechanism. The system has two synchronization states: locked ($\mathit{sync\_locked} = 1$) and unlocked ($\mathit{sync\_locked} = 0$). In the locked state, the VNF monitors the drift of the local clock. If the absolute offset exceeds the wide tolerance limit $\delta_{\mathrm{unlock}}$ designed to ignore transient jitter:}
\begin{equation}
|\mathit{offset}| > \delta_{\mathrm{unlock}}
\end{equation}
\textcolor{blue}{the VNF unlocks the synchronization ($\mathit{sync\_locked} = 0$) to trigger re-synchronization. In the unlocked state, the VNF checks whether the offset has converged within the narrow tolerance limit $\delta_{\mathrm{lock}}$. If $|\mathit{offset}| \le \delta_{\mathrm{lock}}$, the system locks the synchronization state ($\mathit{sync\_locked} = 1$). Otherwise, the VNF decomposes the offset into a slot-level adjustment $S_{\mathrm{adj}}$ and a microsecond-level adjustment $\mu s_{\mathrm{adj}}$ based on the slot duration $T_{\mathrm{slot}}$:}
\begin{equation}
S_{\mathrm{adj}} = \mathit{offset} / T_{\mathrm{slot}}
\end{equation}
\begin{equation}
\mu s_{\mathrm{adj}} = \mathit{offset} \pmod{T_{\mathrm{slot}}}
\end{equation}
\textcolor{blue}{These adjustments are accumulated into the global adjustment variables:}
\begin{equation}
\mathit{slot\_adjustment} \gets \mathit{slot\_adjustment} + S_{\mathrm{adj}}
\end{equation}
\begin{equation}
\mathit{us\_adjustment} \gets \mathit{us\_adjustment} - \mu s_{\mathrm{adj}}
\end{equation}
\textcolor{blue}{The negative sign in the microsecond adjustment reduces the local thread sleep time when the VNF clock is behind, speeding up the slot transition rate to catch up. The synchronization process is summarized in Algorithm~\ref{alg:node_sync_simplified}.}

\begin{algorithm}[H]
\caption{Node Sync}
\label{alg:node_sync_simplified}
\small
\begin{algorithmic}[1]
\Require $t_1, t_2, t_3$ (Timestamps from PNF \textit{UL\_node\_sync}), $t_4$ (VNF local receive timestamp)
\Ensure $\mathit{slot\_adjustment}$ (Slot adjustment accumulator), $\mathit{us\_adjustment}$ (Microsecond adjustment accumulator)

\Statex \textbf{Notation:} 
\Statex $\mathit{offset}$ (Instantaneous timing offset), $\mathit{sync\_locked}$ (Synchronization lock state)
\Statex $\delta_{\mathrm{lock}}$ (Narrow locking threshold), $\delta_{\mathrm{unlock}}$ (Wide unlocking threshold)
\Statex $T_{\mathrm{slot}}$ (Slot duration in \textmu s), $T_{\mathrm{wrap}}$ (Wrap-around period, $10240000$ \textmu s)
\Statex $\text{WrapCorrection}(x, T)$ (Corrects $x$ within $[-T/2, T/2]$)

\Statex \textbf{Phase I: Timing Offset Calculation}
\State $d_1 \gets \text{WrapCorrection}(t_2 - t_1, T_{\mathrm{wrap}})$
\State $d_2 \gets \text{WrapCorrection}(t_4 - t_3, T_{\mathrm{wrap}})$
\State $\mathit{offset} \gets (d_1 - d_2) / 2$

\Statex \textbf{Phase II: Hysteresis Lock and Clock Adjustment}
\If{$\mathit{sync\_locked} = 1 \land |\mathit{offset}| > \delta_{\mathrm{unlock}}$}
    \State $\mathit{sync\_locked} \gets 0$ \Comment{Unlock due to excessive clock drift}
\EndIf

\If{$\mathit{sync\_locked} = 0$}
    \If{$|\mathit{offset}| \le \delta_{\mathrm{lock}}$}
        \State $\mathit{sync\_locked} \gets 1$ \Comment{Lock synchronization}
    \Else
        \State $S_{\mathrm{adj}} \gets \mathit{offset} / T_{\mathrm{slot}}$ \Comment{Integer division}
        \State $\mu s_{\mathrm{adj}} \gets \mathit{offset} \pmod{T_{\mathrm{slot}}}$
        \State $\mathit{slot\_adjustment} \gets \mathit{slot\_adjustment} + S_{\mathrm{adj}}$
        \State $\mathit{us\_adjustment} \gets \mathit{us\_adjustment} - \mu s_{\mathrm{adj}}$
    \EndIf
\EndIf
\State \Return $\mathit{slot\_adjustment}, \mathit{us\_adjustment}$
\end{algorithmic}
\end{algorithm}

\section{Outer-Loop: Delay Management}
\label{subsec:delay_management}

\textcolor{blue}{To counteract changing network latency and maintain scheduling stability, the outer loop implements a dynamic delay management strategy. This strategy operates as an Asymmetric Feedback Controller based on a Risk-Aware Predictive Model, incorporating standard telecommunication jitter measurements to calculate and manage risk debt.}

\subsection{Standardized Interarrival Jitter Calculation}
\label{subsubsec:rfc3550}
\textcolor{blue}{Rather than using an ad-hoc smoothing model, the outer loop strictly implements the standardized interarrival jitter calculation specified in IETF RFC 3550 (RTP) Section 6.4.1. Let $R_i$ be the arrival timestamp of packet $i$ at the physical layer (PNF), and $S_i$ be the transmit timestamp extracted from the nFAPI P7 header of packet $i$. The packet transit time difference $D_i$ between consecutive packets is defined as:}

\begin{equation}
D_i = (R_i - R_{i-1}) - (S_i - S_{i-1})
\end{equation}

\textcolor{blue}{The smoothed interarrival jitter $J_i$ is updated recursively using the Exponentially Weighted Moving Average (EWMA) model:}

\begin{equation}
J_i = J_{i-1} + \frac{|D_i| - J_{i-1}}{16}
\end{equation}

\textcolor{blue}{where the divisor $16$ serves as the smoothing factor, optimizing the trade-off between noise reduction and responsiveness under carrier-grade network conditions. This standardized jitter metric is periodically reported by the PNF within the nFAPI \texttt{Timing Info} message.}

\label{subsubsec:rcgr_control}
\subsection{Asymmetric Predictive Feedback Control}
\textcolor{blue}{Upon receiving the worst-case late arrival offset ($\Delta t_{arrive}$) and the standardized jitter $J_i$ reported by the PNF, the VNF Delay Manager evaluates the predicted tail risk ($Risk_{pred}$) for the next scheduling cycle:}

\begin{equation}
Risk_{pred} = \max(\Delta t_{arrive}, \mu_{delay}) + J_i
\end{equation}

\textcolor{blue}{where $\mu_{delay}$ is the EWMA mean of the late arrival offset. To prevent rapid oscillations, we introduce a history-weighted risk debt ($Debt_{risk}$), which represents a decaying memory of past timing instability:}

\begin{equation}
Debt_{risk} \gets Debt_{risk} + \alpha \cdot (\max(0, Risk_{pred}) - Debt_{risk})
\end{equation}

\textcolor{blue}{The update of the slots ahead $S_{ahead}$ is then governed by the Rapid-Compensation / Gradual-Relaxation (RCGR) asymmetric control law. Under RCGR, the controller reacts instantly to positive risk to preserve connection reliability, while slowly relaxing the timing margin only when the system has remained stable for a sustained duration:}

\begin{itemize}
    \item \textcolor{blue}{\textbf{Rapid Compensation (Preserve Throughput):} If a timing violation is detected ($\Delta t_{arrive} > 0$) or the tail risk is positive ($Risk_{pred} > 0$), the controller immediately increments the slots ahead parameter to prevent scheduling timeouts:}
          \begin{equation}
          S_{next} = \min(S_{curr} + 1, S_{max})
          \end{equation}
          \textcolor{blue}{and resets the stable observation counter $count_{stable} = 0$.}
    \item \textcolor{blue}{\textbf{Gradual Relaxation (Optimize Latency):} If the system is cleared of risk debt ($Debt_{risk} < \epsilon$, where $\epsilon$ is the debt clearance threshold) and has remained stable for a consecutive number of slots ($count_{stable} \ge N_{stable}$), the controller cautiously decrements the advance time to reduce scheduling latency:}
          \begin{equation}
          S_{next} = \max(S_{curr} - 1, 1)
          \end{equation}
          \textcolor{blue}{and resets $count_{stable} = 0$.}
    \item \textcolor{blue}{\textbf{Maintain Baseline:} Otherwise, the controller holds the current timing, $S_{next} = S_{curr}$, and increments $count_{stable}$ if $Risk_{pred} \le 0$.}
\end{itemize}

The core adjustment strategy is condensed in Algorithm~\ref{alg:delay_management}.

\begin{algorithm}[H]
\caption{Delay Management}
\label{alg:delay_management}
\small
\begin{algorithmic}[1]
\Require $\Delta t_{arrive}$ (Observed timing offset), $S_{curr}$ (Current trigger timing)
\Ensure $S_{next}$ (Trigger timing for the next MAC schedule)

\Statex \textbf{Notation:} 
\Statex $\mu_{delay}$ (EWMA mean delay), $J_i$ (RFC 3550 Interarrival Jitter)
\Statex $Risk_{pred}$ (Predicted timing risk), $Debt_{risk}$ (History-weighted risk debt)
\Statex $count_{stable}$ (Consecutive stable arrivals counter)
\Statex $N_{stable}$ (Adaptive stability threshold), $S_{max}$ (Maximum advance-time limit)
\Statex $\alpha$ (Smoothing factor, where $\alpha=1/8$)
\Statex $\epsilon$ (Debt clearance threshold)

\Statex \textbf{Phase 1: Risk and Debt Estimation}
\State $error \gets \Delta t_{arrive} - \mu_{delay}$
\State $\mu_{delay} \gets \mu_{delay} + \alpha \cdot error$
\State $Risk_{pred} \gets \max(\Delta t_{arrive}, \mu_{delay}) + J_i$
\State $Debt_{risk} \gets Debt_{risk} + \alpha \cdot (\max(0, Risk_{pred}) - Debt_{risk})$
\State $IsDebtFree \gets (Debt_{risk} < \epsilon)$

\Statex \textbf{Phase 2: Asymmetric Timing Adjustment (RCGR)}
\If{$\Delta t_{arrive} > 0 \lor Risk_{pred} > 0$} 
    \State $S_{next} \gets \min(S_{curr} + 1, S_{max})$ \Comment{Rapid Compensation}
    \State $count_{stable} \gets 0$
\ElsIf{$IsDebtFree \land (count_{stable} \ge N_{stable})$} 
    \State $S_{next} \gets \max(S_{curr} - 1, 1)$ \Comment{Gradual Relaxation}
    \State $count_{stable} \gets 0$
\Else 
    \State $S_{next} \gets S_{curr}$ \Comment{Maintain timing}
    \If{$Risk_{pred} \le 0$}
        \State $count_{stable} \gets count_{stable} + 1$
    \EndIf
\EndIf
\State \Return $S_{next}$
\end{algorithmic}
\end{algorithm}

\section{Coordination between Inner and Outer Loops}
\label{subsec:loop_coordination}

\textcolor{blue}{To ensure stable operation under real-world conditions, the inner and outer loops must coordinate without inducing control loop oscillations. The microsecond-level Node Sync (Inner-Loop) focuses strictly on correcting clock phase offsets to align the VNF's scheduling thread with the PNF's slot transitions. Because it directly shifts the VNF's time base, any clock adjustment ($S_{adj}, \mu s_{adj}$) temporarily shifts the arrival timestamp reference. The outer loop's Delay Manager (Outer-Loop) is designed to absorb this transient shifting. By relying on the history-weighted risk debt ($Debt_{risk}$), the outer loop avoids reacting to the temporary timing offsets caused by clock adjustments, allowing the inner loop to converge. Once the inner loop locks synchronization ($sync\_locked = 1$), the time base stabilizes, allowing the outer loop to achieve precise slot-level optimizations ($S_{ahead}$).}
