\chapter{Proposed Method}
\label{sec:method}

This chapter details the proposed dynamic timing adjustment framework. To address the unstable transmission latency during packet transmission between the PNF and VNF, we propose two core timing control algorithms: Node Sync for software-level clock synchronization and Delay Management for slot-level adaptive delay management. The timing control of this system is organized as a closed-loop system, ensuring that the VNF-side scheduling can dynamically track and compensate for both microsecond-level clock drift and slot-level transport latency variation. The two core algorithms cooperate as follows:

\begin{enumerate}
    \item \textcolor{blue}{\textbf{Node Sync:} Maintains phase and slot boundary alignment between the VNF and PNF. It periodically exchanges timestamps to calculate the absolute phase offset, correcting for hardware clock drift (Clock Drift).}
    \item \textcolor{blue}{\textbf{Delay Management:} Tracks and estimates packet transport delays and jitter at the slot level, dynamically adjusting the slots ahead ($S_{\mathrm{ahead}}$) to guarantee that control messages fall within the PNF's strict timing window.}
\end{enumerate}

\section{Node Sync}
\label{subsec:node_sync}
\textcolor{blue}{To perform software-level clock synchronization, the system adopts a timestamp exchange mechanism. The VNF periodically transmits a downlink node synchronization message with a local timestamp $t_1$. Upon receiving the message, the PNF records the local arrival timestamp $t_2$ and transmits an uplink node synchronization message back to the VNF, including the transmit timestamp $t_3$ and the echoed $t_2$. When the VNF receives the message, it records its local arrival timestamp $t_4$. Because the nFAPI timestamps are constrained by a 1024 subframe number (SFN) loop, they wrap around every $10.24$~seconds ($10240000$~\textmu s). The algorithm calculates the differences $d_1 = t_2 - t_1$ and $d_2 = t_4 - t_3$, applying a wrap-around correction to align them within the interval $[-5120000, 5120000]$~\textmu s. The VNF then calculates the instantaneous clock offset:}
\begin{equation}
\mathit{offset} = \frac{d_1 - d_2}{2}
\end{equation}
\textcolor{blue}{To prevent timing adjustments from causing control loop oscillations, the synchronization control loop implements a state-based locking mechanism. The system has two synchronization states: locked ($\mathit{sync\_locked} = 1$) and unlocked ($\mathit{sync\_locked} = 0$). In the locked state, the VNF monitors the drift of the local clock. If the absolute offset exceeds the slot duration $T_{\mathrm{slot}}$:}
\begin{equation}
|\mathit{offset}| > T_{\mathrm{slot}}
\end{equation}
\textcolor{blue}{the VNF unlocks the synchronization ($\mathit{sync\_locked} = 0$) to trigger re-synchronization. In the unlocked state, the VNF checks whether the offset has converged within the slot duration $T_{\mathrm{slot}}$. If $|\mathit{offset}| \le T_{\mathrm{slot}}$, the system locks the synchronization state ($\mathit{sync\_locked} = 1$). Otherwise, the VNF decomposes the offset into a slot-level adjustment $S_{\mathrm{adj}}$ and a microsecond-level adjustment $\mu s_{\mathrm{adj}}$ based on the slot duration $T_{\mathrm{slot}}$:}
\begin{equation}
S_{\mathrm{adj}} = \mathit{offset} / T_{\mathrm{slot}}
\end{equation}
\begin{equation}
\mu s_{\mathrm{adj}} = \mathit{offset} \pmod{T_{\mathrm{slot}}}
\end{equation}
\textcolor{blue}{These adjustments are accumulated into the global adjustment variables:}
\begin{equation}
\mathit{slot\_adjustment} \gets \mathit{slot\_adjustment} + S_{\mathrm{adj}}
\end{equation}
\begin{equation}
\mathit{us\_adjustment} \gets \mathit{us\_adjustment} - \mu s_{\mathrm{adj}}
\end{equation}
\textcolor{blue}{The negative sign in the microsecond adjustment reduces the local thread sleep time when the VNF clock is behind, speeding up the slot transition rate to catch up. The synchronization process is summarized in Algorithm~\ref{alg:node_sync_simplified}.}

\begin{algorithm}[H]
\caption{Node Sync}
\label{alg:node_sync_simplified}
\small
\begin{algorithmic}[1]
\algrenewcommand\algorithmicrequire{\textbf{Input:}}
\algrenewcommand\algorithmicensure{\textbf{Output:}}
\Require $t_1, t_2, t_3$ (Timestamps from PNF \textit{UL\_node\_sync}), $t_4$ (VNF local receive timestamp)
\Ensure $\mathit{slot\_adjustment}$ (Slot adjustment accumulator), $\mathit{us\_adjustment}$ (Microsecond adjustment accumulator)

\Statex \textbf{Notation:} 
\Statex $\mathit{offset}$ (Instantaneous timing offset), $\mathit{sync\_locked}$ (Synchronization lock state)
\Statex $T_{\mathrm{slot}}$ (Slot duration in \textmu s), $T_{\mathrm{wrap}}$ (Wrap-around period, $10240000$ \textmu s)
\Statex $\text{WrapCorrection}(x, T)$ (Corrects $x$ within $[-T/2, T/2]$)

\Statex \textbf{Phase I: Timing Offset Calculation}
\State $d_1 \gets \text{WrapCorrection}(t_2 - t_1, T_{\mathrm{wrap}})$
\State $d_2 \gets \text{WrapCorrection}(t_4 - t_3, T_{\mathrm{wrap}})$
\State $\mathit{offset} \gets (d_1 - d_2) / 2$

\Statex \textbf{Phase II: Hysteresis Lock and Clock Adjustment}
\If{$\mathit{sync\_locked} = 1 \land |\mathit{offset}| > T_{\mathrm{slot}}$}
    \State $\mathit{sync\_locked} \gets 0$ \Comment{Unlock due to excessive clock drift}
\EndIf

\If{$\mathit{sync\_locked} = 0$}
    \If{$|\mathit{offset}| \le T_{\mathrm{slot}}$}
        \State $\mathit{sync\_locked} \gets 1$ \Comment{Lock synchronization}
    \Else
        \State $S_{\mathrm{adj}} \gets \mathit{offset} / T_{\mathrm{slot}}$ \Comment{Integer division}
        \State $\mu s_{\mathrm{adj}} \gets \mathit{offset} \pmod{T_{\mathrm{slot}}}$
        \State $\mathit{slot\_adjustment} \gets \mathit{slot\_adjustment} + S_{\mathrm{adj}}$
        \State $\mathit{us\_adjustment} \gets \mathit{us\_adjustment} - \mu s_{\mathrm{adj}}$
    \EndIf
\EndIf
\State \Return $\mathit{slot\_adjustment}, \mathit{us\_adjustment}$
\end{algorithmic}
\end{algorithm}

\section{Delay Management}
\label{subsec:delay_management}

\textcolor{blue}{To counteract changing network latency and maintain scheduling stability, a dynamic delay management strategy is implemented. This strategy operates as an Asymmetric Feedback Controller based on a Risk-Aware Predictive Model, periodically evaluating the observed worst-case late arrival offset ($e_{\mathrm{late}}$) reported by the PNF.}

\subsection{EWMA-Based Risk and Uncertainty Estimation}
\label{subsubsec:ewma_estimation}
\textcolor{blue}{To smooth high-frequency network noise while remaining sensitive to timing drift, the VNF Delay Manager maintains a running mean of the late offset ($\mu_{\mathrm{late}}$) and the network jitter ($v_{\mathrm{jitter}}$). Let $e_{\mathrm{late}, i}$ be the observed worst-case late offset in scheduling slot $i$. The EWMA updates for mean latency and jitter are defined as:}
\begin{equation}
\mu_{\mathrm{late}, i} = \mu_{\mathrm{late}, i-1} + \alpha \cdot (e_{\mathrm{late}, i} - \mu_{\mathrm{late}, i-1})
\end{equation}
\begin{equation}
v_{\mathrm{jitter}, i} = v_{\mathrm{jitter}, i-1} + \beta \cdot (|e_{\mathrm{late}, i} - \mu_{\mathrm{late}, i}| - v_{\mathrm{jitter}, i-1})
\end{equation}
\textcolor{blue}{where the smoothing factors are chosen as $\alpha = 1/8$ to track the long-term latency trend, and $\beta = 1/4$ to capture fast jitter fluctuations. These parameters are designed as negative powers of two ($\alpha = 2^{-3}$, $\beta = 2^{-2}$) to allow the controller to perform fast bitwise shift operations in real time.}

\textcolor{blue}{To guide timing adjustments, the controller combines the mean late offset and the estimated jitter to predict the timing risk ($R_{\mathrm{pred}}$). The predicted timing risk for slot $i$ is calculated as the sum of the worst-case boundary of the late offset and the network jitter:}
\begin{equation}
R_{\mathrm{pred}, i} = \max(e_{\mathrm{late}, i}, \mu_{\mathrm{late}, i}) + v_{\mathrm{jitter}, i}
\end{equation}
\textcolor{blue}{Based on this risk estimation, the controller recursively tracks the risk debt ($D_{\mathrm{risk}}$) and the safety margin ($M_{\mathrm{safe}}$):}
\begin{equation}
D_{\mathrm{risk}, i} = D_{\mathrm{risk}, i-1} + \alpha \cdot (\max(0, R_{\mathrm{pred}, i}) - D_{\mathrm{risk}, i-1})
\end{equation}
\begin{equation}
M_{\mathrm{safe}, i} = M_{\mathrm{safe}, i-1} + \alpha \cdot (\max(0, -R_{\mathrm{pred}, i}) - M_{\mathrm{safe}, i-1})
\end{equation}
\textcolor{blue}{The risk debt $D_{\mathrm{risk}}$ aggregates timing violations and predicted risks over time, and the safety margin $M_{\mathrm{safe}}$ tracks the average buffer capacity when the system operates without risk. The system is considered debt-free if $D_{\mathrm{risk}}$ decays to zero ($D_{\mathrm{risk}} \approx 0$).}

\subsection{Asymmetric Predictive Feedback Control}
\label{subsubsec:delay_control}
\textcolor{blue}{The update of the slots ahead trigger timing $S_{\mathrm{next}}$ is governed by an asymmetric control law. Under this law, the scheduler reacts immediately to timing violations or positive risk, while slowly decreasing the lead time only when the system has remained stable for a sustained duration:}
\begin{itemize}
    \item \textcolor{blue}{\textbf{Rapid Compensation (Preserve Throughput):} If a timing violation is detected ($e_{\mathrm{late}} > 0$) or the predicted risk is positive ($R_{\mathrm{pred}} > 0$), the controller immediately increments the slots ahead parameter to prevent scheduling timeouts:}
          \begin{equation}
          S_{\mathrm{next}} = \min\left(S_{\mathrm{curr}} + 1, \left\lfloor \frac{T_{\mathrm{window}}}{T_{\mathrm{slot}}} \right\rfloor\right)
          \end{equation}
          \textcolor{blue}{and resets the stable observation counter $C_{\mathrm{stable}} = 0$.}
    \item \textcolor{blue}{\textbf{Gradual Relaxation (Optimize Latency):} If the system is cleared of risk debt ($D_{\mathrm{risk}} = 0$), the current sample has no risk ($R_{\mathrm{pred}} \le 0$), and the safety margin is sufficient ($M_{\mathrm{safe}} \ge v_{\mathrm{jitter}}$), the controller decrements the slots ahead timing once the stability condition is satisfied:}
          \begin{equation}
          S_{\mathrm{next}} = \max(S_{\mathrm{curr}} - 1, 1)
          \end{equation}
          \textcolor{blue}{and resets $C_{\mathrm{stable}} = 0$. To prevent premature timing reductions as the scheduling buffer grows, the required stable count threshold increases quadratically with $S_{\mathrm{curr}}$.}
    \item \textcolor{blue}{\textbf{Maintain Baseline:} Otherwise, the controller holds the current timing, $S_{\mathrm{next}} = S_{\mathrm{curr}}$. The stable counter $C_{\mathrm{stable}}$ is incremented if the current sample is safe ($R_{\mathrm{pred}} \le 0$).}
\end{itemize}

The core adjustment strategy is condensed in Algorithm~\ref{alg:delay_management}.

\begin{algorithm}[!htbp]
\caption{Delay Management}
\label{alg:delay_management}
\footnotesize
\begin{algorithmic}[1]
\algrenewcommand\algorithmicrequire{\textbf{Input:}}
\algrenewcommand\algorithmicensure{\textbf{Output:}}
\Require $e_{\mathrm{late}}$ (Worst-case observed late offset), $S_{\mathrm{curr}}$ (Current trigger timing),
\Statex \hskip\algorithmicindent $T_{\mathrm{window}}$ (Timing window), $T_{\mathrm{slot}}$ (Slot duration)
\Ensure $S_{\mathrm{next}}$ (Trigger timing for the next MAC schedule)
\Statex 
\Statex \textbf{Notation:} 
\Statex $\mu_{\mathrm{late}}$ (EWMA mean late offset), $v_{\mathrm{jitter}}$ (EWMA network jitter)
\Statex $R_{\mathrm{pred}}$ (Predicted timing risk), $D_{\mathrm{risk}}$ (EWMA risk debt), $M_{\mathrm{safe}}$ (EWMA safety margin)
\Statex $C_{\mathrm{stable}}$ (Stable observation counter)
\Statex $\alpha, \beta$ (Smoothing factors for latency and jitter)
 
\Statex \textbf{Phase I: EWMA Latency and Jitter Estimation}
\State $\mu_{\mathrm{late}} \gets \mu_{\mathrm{late}} + \alpha \cdot (e_{\mathrm{late}} - \mu_{\mathrm{late}})$ \Comment{Update mean late offset}
\State $v_{\mathrm{jitter}} \gets v_{\mathrm{jitter}} + \beta \cdot (|e_{\mathrm{late}} - \mu_{\mathrm{late}}| - v_{\mathrm{jitter}})$ \Comment{Update network jitter}
\State $R_{\mathrm{pred}} \gets \max(e_{\mathrm{late}}, \mu_{\mathrm{late}}) + v_{\mathrm{jitter}}$ \Comment{Evaluate predicted risk}
\State $D_{\mathrm{risk}} \gets D_{\mathrm{risk}} + \alpha \cdot (\max(0, R_{\mathrm{pred}}) - D_{\mathrm{risk}})$ \Comment{Update risk debt}
\State $M_{\mathrm{safe}} \gets M_{\mathrm{safe}} + \alpha \cdot (\max(0, -R_{\mathrm{pred}}) - M_{\mathrm{safe}})$ \Comment{Update safety margin}

\Statex \textbf{Phase II: Asymmetric Timing Adjustment}
\If{$e_{\mathrm{late}} > 0 \lor R_{\mathrm{pred}} > 0$}
    \State $S_{\mathrm{next}} \gets \min(S_{\mathrm{curr}} + 1, \lfloor T_{\mathrm{window}} / T_{\mathrm{slot}} \rfloor)$ \Comment{Rapid Compensation}
    \State $C_{\mathrm{stable}} \gets 0$
\ElsIf{$D_{\mathrm{risk}} \approx 0 \land R_{\mathrm{pred}} \le 0 \land M_{\mathrm{safe}} \ge v_{\mathrm{jitter}}$}
    \If{stability condition is satisfied}
        \State $S_{\mathrm{next}} \gets \max(S_{\mathrm{curr}} - 1, 1)$ \Comment{Gradual Relaxation}
        \State $C_{\mathrm{stable}} \gets 0$
    \Else
        \State $S_{\mathrm{next}} \gets S_{\mathrm{curr}}$
        \State $C_{\mathrm{stable}} \gets C_{\mathrm{stable}} + 1$
    \EndIf
\Else
    \State $S_{\mathrm{next}} \gets S_{\mathrm{curr}}$
    \State $C_{\mathrm{stable}} \gets 0$
\EndIf
\State \Return $S_{\mathrm{next}}$
\end{algorithmic}
\end{algorithm}

\section{Coordination between Node Sync and Delay Management}
\label{subsec:loop_coordination}

\textcolor{blue}{To ensure stable operation under real-world conditions, Node Sync and Delay Management must coordinate without inducing control loop oscillations. The microsecond-level Node Sync focuses strictly on correcting clock phase offsets to align the VNF's scheduling thread with the PNF's slot transitions. Because it directly shifts the VNF's time base, any clock adjustment ($S_{adj}, \mu s_{adj}$) temporarily shifts the arrival timestamp reference. The Delay Manager is designed to absorb this transient shifting. By relying on the history-weighted risk debt ($D_{risk}$), the Delay Manager avoids reacting to the temporary timing offsets caused by clock adjustments, allowing Node Sync to converge. Once Node Sync locks synchronization ($sync\_locked = 1$), the time base stabilizes, allowing the Delay Manager to achieve precise slot-level optimizations ($S_{\mathrm{ahead}}$).}
