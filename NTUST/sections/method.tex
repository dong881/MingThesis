This chapter details the proposed dynamic timing adjustment algorithm. To address the unstable transmission latency during packet transmission between the PNF and VNF, we propose an Adaptive Slots Ahead mechanism.

\begin{enumerate}
    \item \textcolor{blue}{\textbf{Inner-Loop (Node Sync):} Maintains phase and slot boundary alignment between the VNF and PNF. It periodically exchanges timestamps to calculate the absolute phase offset, correcting for hardware clock drift (Clock Drift).}
    \item \textcolor{blue}{\textbf{Outer-Loop (Delay Management):} Operates on top of the stabilized phase reference provided by the inner loop. It tracks and estimates packet transport delays and jitter at the slot level, dynamically adjusting the slots ahead ($S_{ahead}$) to guarantee that control messages fall within the PNF's strict timing window.}
\end{enumerate}

\section{Inner-Loop: Node Sync}
\label{subsec:node_sync}
\textcolor{blue}{To perform software-level clock synchronization, the system adopts a timestamp exchange mechanism similar to the Network Time Protocol (NTP). The VNF periodically transmits a downlink node synchronization message with a local timestamp $t_1$. Upon receiving the message, the PNF records the local arrival timestamp $t_2$ and transmits an uplink node synchronization message back to the VNF, including the transmit timestamp $t_3$ and the echoed $t_2$. When the VNF receives the message, it records its local arrival timestamp $t_4$. Because the nFAPI timestamps are constrained by a 1024 subframe number (SFN) loop, they wrap around every $10.24$~seconds ($10240000$~\textmu s). The algorithm calculates the differences $d_1 = t_2 - t_1$ and $d_2 = t_4 - t_3$, applying a wrap-around correction to align them within the interval $[-5120000, 5120000]$~\textmu s. The VNF then calculates the instantaneous clock offset:}
\begin{equation}
\mathit{offset} = \frac{d_1 - d_2}{2}
\end{equation}
\textcolor{blue}{To prevent timing adjustments from causing control loop oscillations, the synchronization control loop implements a state-based dual-threshold hysteresis mechanism. The system has two synchronization states: locked ($\mathit{sync\_locked} = 1$) and unlocked ($\mathit{sync\_locked} = 0$). In the locked state, the VNF monitors the drift of the local clock. If the absolute offset exceeds the slot duration $T_{\mathrm{slot}}$:}
\begin{equation}
|\mathit{offset}| > T_{\mathrm{slot}}
\end{equation}
\textcolor{blue}{the VNF unlocks the synchronization ($\mathit{sync\_locked} = 0$) to trigger re-synchronization. In the unlocked state, the VNF checks whether the offset has converged within the synchronization locking tolerance limit $\delta$. If $|\mathit{offset}| \le \delta$, the system locks the synchronization state ($\mathit{sync\_locked} = 1$). Otherwise, the VNF decomposes the offset into a slot-level adjustment $S_{\mathrm{adj}}$ and a microsecond-level adjustment $\mu s_{\mathrm{adj}}$ based on the slot duration $T_{\mathrm{slot}}$:}
\begin{equation}
S_{\mathrm{adj}} = \mathit{offset} / T_{\mathrm{slot}}
\end{equation}
\begin{equation}
\mu s_{\mathrm{adj}} = \mathit{offset} \pmod{T_{\mathrm{slot}}}
\end{equation}
\textcolor{blue}{These adjustments are accumulated into the global adjustment variables:}
\begin{equation}
\mathit{slot\_adjustment} \gets \mathit{slot\_adjustment} + S_{\mathrm{adj}}
\end{equation}
\begin{equation}
\mathit{us\_adjustment} \gets \mathit{us\_adjustment} - \mu s_{\mathrm{adj}}
\end{equation}
\textcolor{blue}{The negative sign in the microsecond adjustment reduces the local thread sleep time when the VNF clock is behind, speeding up the slot transition rate to catch up. The synchronization process is summarized in Algorithm~\ref{alg:node_sync_simplified}.}

\begin{algorithm}[H]
\caption{Node Sync}
\label{alg:node_sync_simplified}
\small
\begin{algorithmic}[1]
\Require $t_1, t_2, t_3$ (Timestamps from PNF \textit{UL\_node\_sync}), $t_4$ (VNF local receive timestamp)
\Ensure $\mathit{slot\_adjustment}$ (Slot adjustment accumulator), $\mathit{us\_adjustment}$ (Microsecond adjustment accumulator)

\Statex \textbf{Notation:} 
\Statex $\mathit{offset}$ (Instantaneous timing offset), $\mathit{sync\_locked}$ (Synchronization lock state)
\Statex $\delta$ (Synchronization locking tolerance)
\Statex $T_{\mathrm{slot}}$ (Slot duration in \textmu s), $T_{\mathrm{wrap}}$ (Wrap-around period, $10240000$ \textmu s)
\Statex $\text{WrapCorrection}(x, T)$ (Corrects $x$ within $[-T/2, T/2]$)

\Statex \textbf{Phase I: Timing Offset Calculation}
\State $d_1 \gets \text{WrapCorrection}(t_2 - t_1, T_{\mathrm{wrap}})$
\State $d_2 \gets \text{WrapCorrection}(t_4 - t_3, T_{\mathrm{wrap}})$
\State $\mathit{offset} \gets (d_1 - d_2) / 2$

\Statex \textbf{Phase II: Hysteresis Lock and Clock Adjustment}
\If{$\mathit{sync\_locked} = 1 \land |\mathit{offset}| > T_{\mathrm{slot}}$}
    \State $\mathit{sync\_locked} \gets 0$ \Comment{Unlock due to excessive clock drift}
\EndIf

\If{$\mathit{sync\_locked} = 0$}
    \If{$|\mathit{offset}| \le \delta$}
        \State $\mathit{sync\_locked} \gets 1$ \Comment{Lock synchronization}
    \Else
        \State $S_{\mathrm{adj}} \gets \mathit{offset} / T_{\mathrm{slot}}$ \Comment{Integer division}
        \State $\mu s_{\mathrm{adj}} \gets \mathit{offset} \pmod{T_{\mathrm{slot}}}$
        \State $\mathit{slot\_adjustment} \gets \mathit{slot\_adjustment} + S_{\mathrm{adj}}$
        \State $\mathit{us\_adjustment} \gets \mathit{us\_adjustment} - \mu s_{\mathrm{adj}}$
    \EndIf
\EndIf
\State \Return $\mathit{slot\_adjustment}, \mathit{us\_adjustment}$
\end{algorithmic}
\end{algorithm}

\section{Outer-Loop: Delay Management}
\label{subsec:delay_management}

\textcolor{blue}{To counteract changing network latency and maintain scheduling stability, the outer loop implements a dynamic delay management strategy. This strategy operates as an Asymmetric Feedback Controller based on a Risk-Aware Predictive Model, incorporating standard telecommunication jitter measurements to calculate and manage risk debt.}

\subsection{EWMA-Based Risk and Uncertainty Estimation}
\label{subsubsec:ewma_estimation}
\textcolor{blue}{To smooth high-frequency network noise while remaining sensitive to timing drift, the VNF Delay Manager maintains a running mean of the late offset ($\mu_{\mathrm{late}}$) and its directional jitter deviation ($J_{\mathrm{late}}$) along with the variance ($v_{\mathrm{jitter}}$). Let $e_{\mathrm{late}, i}$ be the observed worst-case late offset in scheduling window $i$. The EWMA updates are defined as:}
\begin{equation}
\mu_{\mathrm{late}, i} = \mu_{\mathrm{late}, i-1} + \alpha \cdot (e_{\mathrm{late}, i} - \mu_{\mathrm{late}, i-1})
\end{equation}
\begin{equation}
v_{\mathrm{jitter}, i} = v_{\mathrm{jitter}, i-1} + \beta \cdot (|e_{\mathrm{late}, i} - \mu_{\mathrm{late}, i}| - v_{\mathrm{jitter}, i-1})
\end{equation}
\textcolor{blue}{where the smoothing factors are chosen as $\alpha = 1/8$ and $\beta = 1/4$. These parameters are specifically designed as negative powers of two ($\alpha = 2^{-3}$, $\beta = 2^{-2}$) to allow the controller to perform fast bitwise shift operations in real time while maintaining an optimal balance between estimation stability and tracking sensitivity.
The timing uncertainty is then calculated as the sum of the directional jitter and the average variation:}
\begin{equation}
U_{\mathrm{timing}} = J_{\mathrm{late}} + v_{\mathrm{jitter}}
\end{equation}
\textcolor{blue}{where $J_{\mathrm{late}}$ is updated recursively using the factor $\beta$ on the positive (late) or negative (early) offset deviations. Using this uncertainty, the predicted risk ($Risk_{\mathrm{pred}}$) and the safety margin ($M_{\mathrm{sample}}$) are evaluated:}
\begin{equation}
Risk_{\mathrm{pred}} = \max(e_{\mathrm{late}}, \mu_{\mathrm{late}}) + U_{\mathrm{timing}}
\end{equation}
\begin{equation}
M_{\mathrm{sample}} = \max(0, -Risk_{\mathrm{pred}})
\end{equation}
\textcolor{blue}{The long-term risk debt ($Debt_{\mathrm{risk}}$) and the average safety margin ($M_{\mathrm{safe\_ewma}}$) are updated recursively using the factor $\alpha$. The system is considered debt-free if $Debt_{\mathrm{risk}}$ decays to zero.}

\subsection{Asymmetric Predictive Feedback Control}
\label{subsubsec:delay_control}
\textcolor{blue}{The update of the slots ahead trigger timing $S_{next}$ is governed by an asymmetric control law. Under this law, the scheduler reacts immediately to timing violations or positive risk, while slowly decreasing the lead time only when the system has remained stable for a sustained duration:}
\begin{itemize}
    \item \textcolor{blue}{\textbf{Rapid Compensation (Preserve Throughput):} If a timing violation is detected ($e_{\mathrm{late}} > 0$) or the predicted risk is positive ($Risk_{\mathrm{pred}} > 0$), the controller immediately increments the slots ahead parameter to prevent scheduling timeouts:}
          \begin{equation}
          S_{next} = \min(S_{curr} + 1, S_{max})
          \end{equation}
          \textcolor{blue}{and resets the stable observation counter $count_{stable} = 0$.}
    \item \textcolor{blue}{\textbf{Gradual Relaxation (Optimize Latency):} If the system is cleared of risk debt ($Debt_{\mathrm{risk}} = 0$), the current sample has no risk, and the safety margin is sufficient ($M_{\mathrm{safe\_ewma}} \ge U_{\mathrm{timing}}$), the controller decrements the slots ahead timing once the stability condition is satisfied:}
          \begin{equation}
          S_{next} = \max(S_{curr} - 1, 1)
          \end{equation}
          \textcolor{blue}{and resets $count_{stable} = 0$. To prevent premature timing reductions as the scheduling buffer grows, the required stable count threshold increases quadratically with $S_{curr}$.}
    \item \textcolor{blue}{\textbf{Maintain Baseline:} Otherwise, the controller holds the current timing, $S_{next} = S_{curr}$. The stable counter $count_{stable}$ is incremented if the current sample is safe.}
\end{itemize}

The core adjustment strategy is condensed in Algorithm~\ref{alg:delay_management}.

\begin{algorithm}[H]
\caption{Delay Management}
\label{alg:delay_management}
\small
\begin{algorithmic}[1]
\Require $e_{\mathrm{late}}$ (Worst-case observed late offset), $S_{curr}$ (Current trigger timing)
\Ensure $S_{next}$ (Trigger timing for the next MAC schedule)

\Statex \textbf{Notation:} 
\Statex $\mu_{\mathrm{late}}$ (EWMA mean late offset), $v_{\mathrm{jitter}}$ (EWMA jitter variance)
\Statex $U_{\mathrm{timing}}$ (Timing uncertainty), $Risk_{\mathrm{pred}}$ (Predicted timing risk)
\Statex $Debt_{\mathrm{risk}}$ (EWMA risk debt), $M_{\mathrm{safe\_ewma}}$ (EWMA safety margin)
\Statex $count_{stable}$ (Stable observation counter)
\Statex $\alpha, \beta$ (Smoothing factors, where $\alpha=1/8, \beta=1/4$)

\Statex \textbf{Phase I: EWMA Risk and Uncertainty Tracking}
\State $error \gets e_{\mathrm{late}} - \mu_{\mathrm{late}}$
\State $\mu_{\mathrm{late}} \gets \mu_{\mathrm{late}} + \alpha \cdot error$
\State $v_{\mathrm{jitter}} \gets v_{\mathrm{jitter}} + \beta \cdot (|error| - v_{\mathrm{jitter}})$
\State $U_{\mathrm{timing}} \gets \text{CalculateUncertainty}(v_{\mathrm{jitter}}, \beta)$
\State $Risk_{\mathrm{pred}} \gets \max(e_{\mathrm{late}}, \mu_{\mathrm{late}}) + U_{\mathrm{timing}}$
\State $Debt_{\mathrm{risk}} \gets Debt_{\mathrm{risk}} + \alpha \cdot (\max(0, Risk_{\mathrm{pred}}) - Debt_{\mathrm{risk}})$
\State $M_{\mathrm{sample}} \gets \max(0, -Risk_{\mathrm{pred}})$
\State $M_{\mathrm{safe\_ewma}} \gets M_{\mathrm{safe\_ewma}} + \alpha \cdot (M_{\mathrm{sample}} - M_{\mathrm{safe\_ewma}})$

\Statex \textbf{Phase II: Asymmetric Timing Adjustment}
\If{$e_{\mathrm{late}} > 0 \lor Risk_{\mathrm{pred}} > 0$}
    \State $S_{next} \gets \min(S_{curr} + 1, S_{max})$ \Comment{Rapid Compensation}
    \State $count_{stable} \gets 0$
\ElsIf{$Debt_{\mathrm{risk}} \approx 0 \land e_{\mathrm{late}} \le 0 \land M_{\mathrm{safe\_ewma}} \ge U_{\mathrm{timing}}$}
    \If{stability condition is satisfied}
        \State $S_{next} \gets \max(S_{curr} - 1, 1)$ \Comment{Gradual Relaxation}
        \State $count_{stable} \gets 0$
    \Else
        \State $S_{next} \gets S_{curr}$
        \State $count_{stable} \gets count_{stable} + 1$
    \EndIf
\Else
    \State $S_{next} \gets S_{curr}$
    \State $count_{stable} \gets 0$
\EndIf
\State \Return $S_{next}$
\end{algorithmic}
\end{algorithm}

\section{Coordination between Inner and Outer Loops}
\label{subsec:loop_coordination}

\textcolor{blue}{To ensure stable operation under real-world conditions, the inner and outer loops must coordinate without inducing control loop oscillations. The microsecond-level Node Sync (Inner-Loop) focuses strictly on correcting clock phase offsets to align the VNF's scheduling thread with the PNF's slot transitions. Because it directly shifts the VNF's time base, any clock adjustment ($S_{adj}, \mu s_{adj}$) temporarily shifts the arrival timestamp reference. The outer loop's Delay Manager (Outer-Loop) is designed to absorb this transient shifting. By relying on the history-weighted risk debt ($Debt_{risk}$), the outer loop avoids reacting to the temporary timing offsets caused by clock adjustments, allowing the inner loop to converge. Once the inner loop locks synchronization ($sync\_locked = 1$), the time base stabilizes, allowing the outer loop to achieve precise slot-level optimizations ($S_{ahead}$).}
