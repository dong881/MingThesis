\chapter{Proposed Method}
\label{sec:method}

\textcolor{blue}{This section details the proposed dynamic timing adjustment algorithm. To resolve the unstable latency during packet transmission between the PNF and VNF, we propose a Dual-Loop Timing Control Architecture.}

\section{Dual-Loop Architecture}
\label{subsec:dual_loop_architecture}

\textcolor{blue}{The timing control of this system adopts a ``policy-execution separation'' dual-loop architecture, ensuring that the timing advancement on the VNF side can flexibly cope with network jitter without inducing unexpected oscillations:
\begin{enumerate}
    \item \textbf{Convergence Optimization Loop (Policy):} Dynamically analyzes the statistical data reported by the PNF through the convergence optimization algorithm to determine the global target slot-ahead $s_{\text{ahead}}$.
    \item \textbf{Timing Actuator Loop (Execution):} Executed by an independent VNF timing thread. This thread is the only unit in the system capable of altering the absolute sleep time and slot propagation, translating the upper-layer policy target and low-layer microsecond phase deviations into actual timing advancement.
\end{enumerate}}

\section{VNF Timing Actuator}
\label{subsec:vnf_timing_actuator}

\textcolor{blue}{The core function of the \textbf{VNF Timing Actuator} is isolation and precise phase execution. Initially, we define the basic mathematical parameters used in the system. Let $\mu$ denote the subcarrier spacing parameter (Numerology), and the corresponding slot duration (in microseconds) is defined as $T_{\text{slot}}$:
\begin{equation} \label{eq:tslot}
T_{\text{slot}} = \frac{1000}{2^\mu} 
\end{equation}
Let $S_{\text{next}}$ be the target slot-ahead value determined by the policy loop, and $T_{\text{adv}}$ be the accumulated total phase advance in microseconds. For microsecond-level phase correction, let $\Delta_{\text{pending}}$ be the sub-slot phase deviation (in \textmu s). The algorithm calculates the next absolute wake-up time $T_{\text{next}}$:
\begin{equation} \label{eq:tnext}
T_{\text{next}}^{(k)} = T_{\text{next}}^{(k-1)} + T_{\text{slot}} + \Delta_{\text{pending}}
\end{equation}
Before receiving the initial timing anchor from the PNF, the actuator thread blocks execution via a condition variable. Once active, the thread operates within a mutex-locked environment and triggers sleep according to the absolute monotonic clock calculated in \eqref{eq:tnext}, mitigating long-term error accumulation. By adjusting the absolute wake-up time, it translates the upper-layer policy target ($S_{\text{next}}$) and low-layer microsecond phase deviations into actual timing advancement.}

\section{Convergence Optimization Policy}
\label{subsec:convergence_optimization}

\textcolor{blue}{To counteract fluctuating network latency and maintain scheduling robustness, this paper proposes a dynamic convergence optimization policy. Upon receiving the timing statistics reported by the PNF, the system incorporates an Exponentially Weighted Moving Average (EWMA) model to estimate the mean latency $mean$ and jitter variance $var$. Given the worst-case late arrival $t_{late}$ in the current observation period, the iterative update formulas are:
\begin{align}
\label{eq:ewma_mean}
mean^{(k)} ={}& mean^{(k-1)} + \frac{t_{late} - mean^{(k-1)}}{8} \\
\label{eq:ewma_var}
var^{(k)} ={}& var^{(k-1)} + \frac{|t_{late} - mean^{(k-1)}| - var^{(k-1)}}{4}
\end{align}}

\textcolor{blue}{Before making tracking adjustments, a dynamic safety margin $Safe\_Boundary$ is established to cover the networking jitter. Furthermore, the algorithm designs multiple \textbf{Panic Criteria} triggers. A statistical anomaly is detected if the absolute delay deviation $|t_{late} - mean|$ exceeds $3 \times var$, indicating a sudden latency spike. A strict violation occurs if $t_{late} \ge T_{\text{slot}}$, meaning the packet is late beyond the slot boundary. Either condition triggers a \textbf{Panic Mode}.}

\textcolor{blue}{To prevent unexpected oscillation during target transitions, a \textbf{Leaky Bucket Counter} mechanism acts as an anti-bounce filter. It filters out pseudo-stable low-latency states to prevent prematurely lowering the slots ahead. When panic events are detected, the algorithm initiates a \textbf{Fast up} mode, radically increasing $S_{\text{next}}$ by $N_{jump}$ to ensure connection continuity and resetting the penalty threshold. Conversely, when redundant phase advance exists continuously and the latency is well below the safety boundary ($t_{late} \le -Safe\_Boundary$), the leaky bucket counter decays. Once it reaches zero, a \textbf{Proactive Slow Decay} backoff is triggered to minimize the round-trip time without sacrificing stability. The unified dynamic adjustment is condensed in Algorithm~\ref{alg:delay_manager_ewma}.}

\begin{algorithm}[H]
\caption{Delay Manager: Statistical Variance-Based Control}
\label{alg:delay_manager_ewma}
\begin{algorithmic}[1]
\Require 
    \Statex $t_{late}$: The latest arriving timing offset from PNF
    \Statex $S_{curr}$: Current slots ahead (trigger timing)
\Ensure 
    \Statex $S_{next}$: Proposed slots ahead for the next MAC scheduler

\Statex \textbf{/* Phase 1: EWMA Estimation */}
\State $Delay\_Deviation \gets t_{late} - mean$
\State $mean \gets mean + (Delay\_Deviation / 8)$
\State $var \gets var + (|Delay\_Deviation| - var) / 4$

\Statex \textbf{/* Phase 2: Detect "Near out of Window" */}
\State $Statistical\_Anomaly \gets |Delay\_Deviation| > (3 \times var)$
\State $Panic\_Mode \gets Statistical\_Anomaly \textbf{ or } (t_{late} \ge T_{slot})$

\Statex \textbf{/* Phase 3: Fast up \& Proactive Slow Decay */}
\If{$Panic\_Mode$}
    \State $S_{next} \gets S_{curr} + N_{jump}$ \Comment{\textbf{Fast up:} Jump slots to ensure survival}
    \State $Leaky\_Bucket\_Counter \gets Penalty\_Threshold$ \Comment{Reset decay counter}
\ElsIf{$t_{late} \le -Safe\_Boundary$}
    \If{$Leaky\_Bucket\_Counter > 0$}
        \State $Leaky\_Bucket\_Counter \gets Leaky\_Bucket\_Counter - 1$ \Comment{Stable period decay}
        \State $S_{next} \gets S_{curr}$ \Comment{Prevent ping-pong effect}
    \Else
        \State $S_{next} \gets S_{curr} - 1$ \Comment{\textbf{Proactive Slow Decay:} Minimize RTT}
        \State $Leaky\_Bucket\_Counter \gets Normal\_Threshold$
    \EndIf
\Else
    \State $S_{next} \gets S_{curr}$ \Comment{Maintain current timing}
\EndIf

\State $S_{next} \gets \max(1, \min(S_{next}, Max\_Slots))$ \Comment{Enforce system bounds}
\State \Return $S_{next}$
\end{algorithmic}
\end{algorithm}
